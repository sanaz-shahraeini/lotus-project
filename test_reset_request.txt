from Alvand.models import Users, PasswordResetRequest

# Find a test user
test_user = Users.objects.exclude(groupname='supporter').first()

if test_user:
    # Create password reset request
    PasswordResetRequest.objects.create(user=test_user)
    print(f"✅ Created reset request for: {test_user.username}")
else:
    print("❌ No user found")

# Check all reset requests
print(f"\nTotal reset requests: {PasswordResetRequest.objects.filter(resolved=False).count()}")
for req in PasswordResetRequest.objects.filter(resolved=False):
    print(f"  - {req.user.username}: {req.created_at}")
